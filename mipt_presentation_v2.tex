\begin{frame}{Максимальная дальность}
\justifying
Интенсивность звука в атмосфере затухает экспоненциально:
\begin{equation*}
  I = I_0 e^{-\alpha r}, \text{ где}
  \end{equation*}
  {\footnotesize
  \begin{align*}
  \alpha &=
  8.686 \, f^2
  \Biggl[
  \frac{1.84 \times 10^{-11}}{p}
  \sqrt{\frac{T}{293.15}}
  \\[0.5em]
  &\quad+
  \left(\frac{T}{293.15}\right)^{-5/2}
  \left(
  \frac{0.01275 e^{-2239.1/T}}{f_{r,\mathrm{O}} + f^2/f_{r,\mathrm{O}}}
  +
  \frac{0.1068 e^{-3352/T}}{f_{r,\mathrm{N}} + f^2/f_{r,\mathrm{N}}}
  \right)
  \Biggr],\text{ где}
  \end{align*}
  }
  {\footnotesize
  \begin{itemize}
  \item $r$ - расстояние, м;
  \item $f$ — частота звука, Гц;
  \item $T$ — абсолютная температура, К;
  \item $p$ — атмосферное давление, кПа;
  \item $f_{r,\mathrm{O}}$, $f_{r,\mathrm{N}}$ — частоты релаксации кислорода и азота.
  \end{itemize}
  }

\end{frame}

\begin{frame}{Точность измерения расстояния}
\justifying
Точность измерения дистанции определяется точностью измерений времени:
\begin{equation*}
\sigma_d = c\Delta t.
\end{equation*}

\smallskip

Характерные значения:

\end{frame}


\begin{frame}{Скорость звука во влажном воздухе}
\justifying
Скорость звука в смеси сухого воздуха и водяного пара:
\begin{equation*}
c = \sqrt{\gamma\, R_{\mathrm{mix}}\, T},
\end{equation*}

\smallskip

\begin{itemize}
\item $\gamma = c_{p,\mathrm{mix}} / c_{v,\mathrm{mix}}$ — показатель адиабаты смеси;
\item $R_{\mathrm{mix}} = R / M_{\mathrm{mix}}$ — удельная газовая постоянная;
\item $M_{\mathrm{mix}} = (1 - x_v) M_d + x_v M_v$ — молярная масса смеси;
\item $x_v = p_v / p$ — мольная доля водяного пара;
\item $p_v = h\, p_{\mathrm{sat}}(T)$ — парциальное давление водяного пара;
\item $T$ — абсолютная температура.
\end{itemize}

\end{frame}


\begin{frame}{Погрешность скорости звука}
\justifying
Полная погрешность скорости:
\begin{equation*}
\Delta c = \sqrt{\left(\tfrac{\partial c}{\partial T}\Delta T\right)^2
+ \left(\tfrac{\partial c}{\partial p}\Delta p\right)^2
+ \left(\tfrac{\partial c}{\partial h}\Delta h\right)^2
+ \Delta_{\text{мод}}^2}.
\end{equation*}

\smallskip

При типичных датчиках:
\begin{itemize}
\item $\Delta T = \pm 0{,}5\,^\circ$C $\;\Rightarrow\;\Delta c_T \approx 0{,}30$~м/с;
\item $\Delta p = \pm 120$~Па $\;\Rightarrow\;\Delta c_p \approx 0{,}06$~м/с;
\item $\Delta h = \pm 3\%$ $\;\Rightarrow\;\Delta c_h \approx 0{,}002$~м/с;
\item $\Delta_{\text{мод}} \approx 0{,}31$~м/с.
\end{itemize}

\bigskip

{\bf Итого:} $\Delta c \approx 0{,}43$~м/с $(\pm 0{,}13\%)$.

\end{frame}

\begin{frame}{GDOP — аналитическая формула}
\justifying

Для четырёх якорей в вершинах квадрата со стороной $a$ и объекта на высоте $H$:

\begin{equation*}
D = \sqrt{\tfrac{a^2}{2} + H^2} \quad \text{— расстояние до якоря}.
\end{equation*}

\begin{block}{}
\centering
\boxed{\;\sigma_{xyz} = \sigma_d \cdot D \cdot \sqrt{\tfrac{2}{a^2} + \tfrac{1}{4H^2}}\;}
\end{block}

\bigskip

\begin{itemize}
\item $\sigma_{xyz}$ — СКО положения в 3D;
\item $\sigma_d$ — СКО измерения расстояния;
\item $a$ — сторона квадрата якорей;
\item $H$ — высота объекта.
\end{itemize}

\end{frame}

\begin{frame}{Метод Гаусса--Ньютона}
\justifying

Итерационная минимизация суммы взвешенных квадратов невязок:

\begin{equation*}
\boxed{\;\min_{\mathbf{p}}\; J(\mathbf{p}),
\qquad
J(\mathbf{p}) = \sum_{i=1}^{N} w_i \bigl(\hat r_i - \|\mathbf{p} - \mathbf{a}_i\|\bigr)^2\;}
\end{equation*}

\bigskip

\begin{itemize}
\item $\hat r_i$ — измеренное расстояние от $i$-го якоря до объекта;
\item $\mathbf{a}_i$ — координаты $i$-го якоря;
\item $\mathbf{p}$ — искомые координаты объекта;
\item $w_i$ — вес измерения (обычно обратный дисперсии);
\item $J(\mathbf{p})$ — целевая функция.
\end{itemize}

\end{frame}


\begin{frame}{Аналитический метод}
\justifying
Координаты точки $P$ удовлетворяют системе уравнений трёх сфер:
\begin{equation*}
(x - x_i)^2 + (y - y_i)^2 + z^2 = r_i^2, \qquad i = 1, 2, 3.
\end{equation*}

\smallskip

Вычитая первое уравнение, получаем линейную систему по $(x, y)$:
\begin{align*}
2(x_2 - x_1)\,x + 2(y_2 - y_1)\,y &= r_1^2 - r_2^2 + x_2^2 - x_1^2 + y_2^2 - y_1^2, \\
2(x_3 - x_1)\,x + 2(y_3 - y_1)\,y &= r_1^2 - r_3^2 + x_3^2 - x_1^2 + y_3^2 - y_1^2.
\end{align*}

\smallskip

В матричной форме:
\begin{equation*}
A \begin{pmatrix} x \\ y \end{pmatrix} = \mathbf{b},
\qquad
\begin{pmatrix} x \\ y \end{pmatrix} = A^{-1} \mathbf{b}.
\end{equation*}

\smallskip

Координата $z$ восстанавливается из одного из исходных уравнений:
\begin{equation*}
z = \pm\sqrt{r_1^2 - (x - x_1)^2 - (y - y_1)^2}.
\end{equation*}

\end{frame}
